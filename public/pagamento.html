function gerarPixBRCode(chave, nome, cidade, valor) { function formatarCampo(id, val) { const texto = String(val); return id + texto.length.toString().padStart(2, '0') + texto; } // Telefone Pix no formato internacional let chaveFormatada = String(chave).replace(/\D/g, ''); if (chaveFormatada.length === 11) { chaveFormatada = '+55' + chaveFormatada; } const merchantAccount = formatarCampo('00', 'br.gov.bcb.pix') + formatarCampo('01', chaveFormatada); const valorStr = Number(valor).toFixed(2); let payload = '000201' + formatarCampo('26', merchantAccount) + '52040000' + '5303986' + formatarCampo('54', valorStr) + '5802BR' + formatarCampo('59', nome.substring(0, 25)) + formatarCampo('60', cidade.substring(0, 15)) + formatarCampo('62', formatarCampo('05', 'BRONZE')) + '6304'; // CRC16-CCITT let crc = 0xFFFF; for (let i = 0; i < payload.length; i++) { crc ^= (payload.charCodeAt(i) << 8); for (let j = 0; j < 8; j++) { if (crc & 0x8000) { crc = ((crc << 1) ^ 0x1021) & 0xFFFF; } else { crc = (crc << 1) & 0xFFFF; } } } const crcHex = crc.toString(16).toUpperCase().padStart(4, '0'); return payload + crcHex; }
